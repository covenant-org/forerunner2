cmake_minimum_required(VERSION 3.20)
project(minio_tool LANGUAGES CXX)

if(NOT miniocpp_DIR)
	set(_miniocpp_candidates)

	if(DEFINED VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
		list(APPEND _miniocpp_candidates "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/miniocpp")
	endif()

	if(DEFINED CMAKE_TOOLCHAIN_FILE)
		get_filename_component(_toolchain_dir "${CMAKE_TOOLCHAIN_FILE}" DIRECTORY)
		get_filename_component(_vcpkg_root "${_toolchain_dir}" DIRECTORY)
		get_filename_component(_vcpkg_root "${_vcpkg_root}" DIRECTORY)
		file(GLOB _miniocpp_glob "${_vcpkg_root}/installed/*/share/miniocpp/miniocpp-config.cmake")
		foreach(_config_path IN LISTS _miniocpp_glob)
			get_filename_component(_config_dir "${_config_path}" DIRECTORY)
			list(APPEND _miniocpp_candidates "${_config_dir}")
		endforeach()
	endif()

	list(REMOVE_DUPLICATES _miniocpp_candidates)

	foreach(_candidate IN LISTS _miniocpp_candidates)
		if(EXISTS "${_candidate}/miniocpp-config.cmake")
			set(miniocpp_DIR "${_candidate}" CACHE PATH "" FORCE)
			break()
		endif()
	endforeach()
endif()

if(miniocpp_DIR)
	get_filename_component(_miniocpp_share "${miniocpp_DIR}" DIRECTORY)
	get_filename_component(_miniocpp_prefix "${_miniocpp_share}" DIRECTORY)
	foreach(_prefix_candidate IN ITEMS "${_miniocpp_share}" "${_miniocpp_prefix}")
		if(_prefix_candidate)
			list(APPEND CMAKE_PREFIX_PATH "${_prefix_candidate}")
		endif()
	endforeach()
	set(Z_VCPKG_OPENSSL_USE_SINGLE_CONFIG 1)
	set(OPENSSL_USE_STATIC_LIBS ON)
	set(OpenSSL_ROOT_DIR "${_miniocpp_prefix}" CACHE PATH "" FORCE)
	set(OPENSSL_ROOT_DIR "${_miniocpp_prefix}" CACHE PATH "" FORCE)
	set(_openssl_dir "${_miniocpp_share}/openssl")
	if(EXISTS "${_openssl_dir}/OpenSSLConfig.cmake")
		set(OpenSSL_DIR "${_openssl_dir}" CACHE PATH "" FORCE)
	endif()
	unset(OPENSSL_INCLUDE_DIR CACHE)
	unset(OPENSSL_CRYPTO_LIBRARY CACHE)
	unset(OPENSSL_SSL_LIBRARY CACHE)
	list(REMOVE_DUPLICATES CMAKE_PREFIX_PATH)
endif()

find_package(miniocpp CONFIG REQUIRED)

add_executable(minio_tool minio.cpp)
target_link_libraries(minio_tool PRIVATE miniocpp::miniocpp)
target_compile_features(minio_tool PRIVATE cxx_std_17)
